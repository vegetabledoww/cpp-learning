    struct LinkNode {
        char value;
        LinkNode* next;
        LinkNode(char c) :value(c), next(nullptr) {}
    };

    LinkNode* creatlinknode(const string & str)
    {
        if (str.empty())return nullptr;
        LinkNode* head = new LinkNode(str[0]);
        LinkNode* cur = head;
        //串起来
        for (size_t i = 0; i < str.length(); i++)
        {
            cur->next = new LinkNode(str[i]);
            cur = cur->next;
        }
        return head;
    }

    char findfirstnode(LinkNode * head1, const string & str2)
    {
        unordered_set<char>charSet(str2.begin(), str2.end());
        LinkNode* cur = head1;
        while (cur)
        {
            if (charSet.find(cur->value) == charSet.end())//当前字符不在str2中
            {
                return cur->value;//将此值返回
            }
            cur = cur->next;
        }
        return '\0';//如果没有找到，返回空字符
    }

    int main()
    {
        string str1, str2;
        cin >> str1 >> str2;
        LinkNode* head1 = creatlinknode(str1);
        char result = findfirstnode(head1, str2);
        if (result != '\0')
            cout << result;
        return 0;

    }
