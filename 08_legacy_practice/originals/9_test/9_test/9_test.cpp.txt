int gcd(int a, int b) {
    return b == 0 ? a : gcd(b, a % b);
}

int maxGcdSubarray(const vector<int>& arr, int n, int k) {
    int maxGcd = -1;
    int currentGcd = 0;
    int start = 0;

    for (int end = 0; end < n; ++end) {
        currentGcd = gcd(currentGcd, arr[end]);
        if (end - start + 1 >= k) {
            if (currentGcd > maxGcd) {
                maxGcd = currentGcd;
            }
            if (currentGcd == arr[start]) {
                ++start;
            }
            else {
                while (end - start + 1 >= k && currentGcd != arr[start]) {
                    currentGcd = gcd(currentGcd, arr[start]);
                    ++start;
                }
            }
        }
    }
    return maxGcd;
}

//0-1 背包问题
int knapsackDP(vector<int>& wgt, vector<int>& val, int cap) {
    int n = wgt.size();
    // 初始化 dp 表
    vector<int>dp(cap + 1, 0);
    // 状态转移
    for (int i = 1; i <= n; i++) {
        for (int c = cap; c >= 1; c--)//倒序进行
        {
            if (wgt[i - 1] <= c)
            {
                dp[c] = max(dp[c],dp[c-wgt[i-1]]+val[i-1]);
            }
        }
    }
    return dp[cap];
}
