//分批函数

int nums(vector<int>&data)
{
    int num = 1;
    //data = { 1,1,2,6,3,100 };//4
    sort(data.begin(), data.end());
    for (int i = 0; i < data.size()-1; i++)
    {
        if (data[i] == data[i + 1])
        {
            continue;
        }
        else
        {
            num++;
        }
    }
    return num;
}
