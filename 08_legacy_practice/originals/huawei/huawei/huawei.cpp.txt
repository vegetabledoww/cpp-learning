//题目：
//有一些不同长度绳子，给定一个期望切割出来条数，求至少达到期望条数能切出来的最大子绳子长度。
//注意：绳子不能拼接。
//下面例子中len表示长度，target表示期望的个数。
//
//例子1：
//	输入：len = { 802, 743, 457, 539 }，target = 11；
//	输出：200；（201的时候没法切出11个绳子）
//
//例子2：
//	输入：len = { 1, 1, 1 }，target = 4;
//	输出：0

bool canCut(const vector<int>& lens, int target, int len)
{
	int count = 0;
	for (int l : lens)
	{
		count += l / len;//切成的子绳子数量
	}
	return count >= target;
}

int maxLength(vector<int>& lens, int target)
{
	if (lens.size() == 0) return 0;
	int left = 1;
	int right = *max_element(lens.begin(), lens.end());
	int result = 0;
	//二分查找的过程
	while (left <= right)
	{
		int mid = left + (right - left) / 2;
		if (canCut(lens, target, mid))
		{
			result = mid;
			left = mid + 1;
		}
		else
		{
			right = mid - 1;
		}
	}
	return result;
}

#include <iostream>
#include <vector>
#include <cmath>

int main() {
    int n, k;
    std::cin >> n >> k;

    std::vector<int> x(n), y(n);

    // Read x coordinates
    for (int i = 0; i < n; ++i) {
        std::cin >> x[i];
    }

    // Read y coordinates
    for (int i = 0; i < n; ++i) {
        std::cin >> y[i];
    }

    int max_count = 0;

    // For each person, count the number of people within distance <= k
    for (int i = 0; i < n; ++i) {
        int count = 0;

        for (int j = 0; j < n; ++j) {
            int distance = std::abs(x[i] - x[j]) + std::abs(y[i] - y[j]);
            if (distance <= k) {
                ++count;
            }
        }

        if (count > max_count) {
            max_count = count;
        }
    }

    std::cout << max_count << std::endl;

    return 0;
}
