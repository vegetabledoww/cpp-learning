#include <iostream>
#include <vector>
#include <algorithm>
#include <cmath>

using namespace std;

// 计算两个数的异或值
int xorOperation(int a, int b) {
    return a ^ b;
}

int main() {
    int n, Q;
    cin >> n >> Q;

    vector<int> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }

    vector<int> l(Q), r(Q), m(Q);
    for (int i = 0; i < Q; ++i) {
        cin >> l[i] >> r[i];
    }
    for (int i = 0; i < Q; ++i) {
        cin >> m[i];
    }

    int totalXor = 0;
    for (int i = 0; i < Q; ++i) {
        int maxVal = 0;
        int prefixXor = 0;
        for (int j = l[i] - 1; j <= r[i] - 1; ++j) {
            prefixXor ^= a[j];
        }
        for (int k = 0; k <= m[i]; ++k) {
            int curVal = prefixXor ^ k;
            maxVal = max(maxVal, curVal);
        }
        totalXor ^= maxVal;
    }

    cout << totalXor << endl;

    return 0;
}
